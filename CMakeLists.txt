cmake_minimum_required(VERSION 3.16)
project(acg_normal_preview LANGUAGES CXX)
add_library(geometry src/geometry.cpp)
target_include_directories(geometry PUBLIC src)
target_compile_features(geometry PUBLIC cxx_std_17)
add_executable(normal_preview src/main.cpp)
target_link_libraries(normal_preview PRIVATE geometry)
include(CTest)
if(BUILD_TESTING)
  add_executable(geometry_tests tests/geometry_tests.cpp)
  target_link_libraries(geometry_tests PRIVATE geometry)
  add_test(NAME geometry COMMAND geometry_tests)
endif()
set_target_properties(normal_preview PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
  target_compile_options(normal_preview PRIVATE /W4 /permissive-)
else()
  target_compile_options(normal_preview PRIVATE -Wall -Wextra -Wpedantic)
endif()

target_link_libraries(scene_library PUBLIC tracer)

get_property(all_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(target IN LISTS all_targets)
  get_target_property(target_type ${target} TYPE)
  if(target_type STREQUAL "UTILITY")
    continue()
  endif()
  set_target_properties(${target} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
